% Pandoc PDF style for Chinese survey reports
% Compatible with: pandoc file.md -o file.pdf --pdf-engine=xelatex -V documentclass=ctexart
\usepackage{indentfirst}
\usepackage{titlesec}
\usepackage{graphicx}
\usepackage{caption}
\usepackage{float}
\usepackage{etoolbox}
\usepackage{fancyhdr}
\usepackage{geometry}

% Font configuration for Windows
\setCJKmainfont[BoldFont={SimHei}, ItalicFont={KaiTi}]{SimSun}
\setCJKsansfont{SimHei}
\setCJKmonofont{FangSong}

% Define font families
\newcommand{\fontheiti}{\bfseries}
\newcommand{\fontkaiti}{\itshape}
\newcommand{\fontfangsong}{\normalfont}
\newcommand{\fontsongti}{\normalfont}

% Body text: Fangsong, size 3 (16pt), fixed leading 28pt
\AtBeginDocument{
  \fontsize{16pt}{28pt}\selectfont
}
\setlength{\parindent}{2em}
\setlength{\parskip}{0pt}

% Keep metadata title only for header; do not print a separate maketitle block
\AtBeginDocument{\let\maketitle\relax}

% Tables: Songti, size 4 (14pt), single line spacing
\AtBeginEnvironment{tabular}{\fontsize{14pt}{14pt}\selectfont\renewcommand{\arraystretch}{1}}
\AtBeginEnvironment{longtable}{\fontsize{14pt}{14pt}\selectfont\renewcommand{\arraystretch}{1}}

% Heading styles:
% #      -> total title: Heiti (Bold), size 2 (22pt)
% ##     -> level-1 title: Heiti (Bold), size 3 (16pt/20pt)
% ###    -> level-2 title: Kaiti (Italic), size 3 (16pt)
% ####   -> level-3 title: Fangsong bold, size 3 (16pt)
\titleformat{\section}
  {\centering\bfseries\fontsize{22pt}{28pt}\selectfont}
  {\thesection}{0.8em}{}
\titleformat{\subsection}
  {\bfseries\fontsize{20pt}{28pt}\selectfont}
  {\thesubsection}{0.6em}{}
\titleformat{\subsubsection}
  {\itshape\fontsize{16pt}{28pt}\selectfont}
  {\thesubsubsection}{0.6em}{}
\titleformat{\paragraph}[hang]
  {\bfseries\fontsize{16pt}{28pt}\selectfont}
  {\theparagraph}{0.6em}{}

\titlespacing*{\section}{0pt}{12pt}{12pt}
\titlespacing*{\subsection}{0pt}{10pt}{8pt}
\titlespacing*{\subsubsection}{0pt}{8pt}{6pt}
\titlespacing*{\paragraph}{0pt}{6pt}{4pt}

% Footer only: centered page numbers, no header rule
\pagestyle{plain}
\raggedbottom
\fancypagestyle{plain}{
  \fancyhf{}
  \fancyfoot[C]{\thepage}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}
}

% Keep inserted figures tidy and attached to their captions
\floatplacement{figure}{htbp}
\renewcommand{\topfraction}{0.9}
\renewcommand{\bottomfraction}{0.8}
\renewcommand{\textfraction}{0.07}
\renewcommand{\floatpagefraction}{0.8}
\setlength{\textfloatsep}{10pt plus 2pt minus 2pt}
\setlength{\intextsep}{10pt plus 2pt minus 2pt}
\setkeys{Gin}{width=0.9\linewidth,keepaspectratio}
\captionsetup[figure]{font={small},justification=centering,singlelinecheck=false,skip=6pt}
